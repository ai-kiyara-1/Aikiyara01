class LocalTransformersAdapter:
    """Optional inference adapter for an existing causal language model.

    This class does not train or create a foundation model. The caller supplies
    a compatible model identifier that is available locally or can be downloaded.
    """

    def __init__(self, model_id: str):
        from transformers import AutoModelForCausalLM, AutoTokenizer

        self.tokenizer = AutoTokenizer.from_pretrained(model_id)
        self.model = AutoModelForCausalLM.from_pretrained(model_id)
        self.model.eval()
        self.device = next(self.model.parameters()).device
        if self.tokenizer.pad_token_id is None:
            self.tokenizer.pad_token = self.tokenizer.eos_token

    def generate(self, system: str, history: list[dict], user_message: str) -> str:
        prompt_parts = [system.strip(), ""]
        for item in history[-8:]:
            role = str(item.get("role", "user")).upper()
            content = str(item.get("content", ""))
            prompt_parts.append(f"{role}: {content}")
        prompt_parts.append(f"USER: {user_message}")
        prompt_parts.append("ASSISTANT:")
        prompt = "\n".join(prompt_parts)

        inputs = self.tokenizer(prompt, return_tensors="pt")
        inputs = {k: v.to(self.device) for k, v in inputs.items()}
        output = self.model.generate(
            **inputs,
            max_new_tokens=160,
            do_sample=True,
            temperature=0.7,
            top_p=0.9,
            pad_token_id=self.tokenizer.pad_token_id,
        )
        generated = output[0][inputs["input_ids"].shape[1]:]
        return self.tokenizer.decode(generated, skip_special_tokens=True).strip()

"""LLM/image provider gateways.

Supports native Gemini REST via GEMINI_API_KEY and OpenAI-compatible APIs as a fallback.
Secrets are read only from server environment variables.
"""
import base64
import json
import os
import urllib.parse
import urllib.request


def _post_json(url, payload, api_key=None, headers=None, timeout=90):
    data = json.dumps(payload).encode("utf-8")
    req = urllib.request.Request(url, data=data, method="POST", headers={"Content-Type": "application/json", **(headers or {})})
    if api_key:
        req.add_header("Authorization", f"Bearer {api_key}")
    with urllib.request.urlopen(req, timeout=timeout) as resp:
        return json.loads(resp.read().decode("utf-8"))


class OpenAICompatibleProvider:
    """Provider facade: native Gemini when GEMINI_API_KEY is set, otherwise OpenAI-compatible."""

    def __init__(self):
        self.gemini_key = os.getenv("GEMINI_API_KEY", "").strip()
        self.gemini_model = os.getenv("GEMINI_MODEL", "gemini-2.5-flash").strip()
        self.base_url = os.getenv("UF_LLM_BASE_URL", "https://api.openai.com/v1").rstrip("/")
        self.api_key = os.getenv("UF_LLM_API_KEY", "").strip()
        self.model = os.getenv("UF_LLM_MODEL", "").strip()

    @property
    def configured(self):
        return bool(self.gemini_key or (self.api_key and self.model))

    @property
    def provider_name(self):
        return "Gemini" if self.gemini_key else "OpenAI-compatible"

    def _gemini_contents(self, messages):
        contents = []
        for item in messages:
            role = "model" if item.get("role") == "assistant" else "user"
            content = item.get("content", "")
            parts = []
            if isinstance(content, str):
                parts.append({"text": content})
            elif isinstance(content, list):
                for part in content:
                    if part.get("type") == "text":
                        parts.append({"text": str(part.get("text", ""))})
                    elif part.get("type") == "image_url":
                        url = part.get("image_url", {}).get("url", "")
                        if url.startswith("data:") and ";base64," in url:
                            header, b64 = url.split(";base64,", 1)
                            mime = header[5:] or "image/jpeg"
                            parts.append({"inline_data": {"mime_type": mime, "data": b64}})
            if parts:
                contents.append({"role": role, "parts": parts})
        return contents

    def _gemini_payload(self, system, messages, temperature=.7, max_tokens=800):
        return {
            "systemInstruction": {"parts": [{"text": system}]},
            "contents": self._gemini_contents(messages),
            "generationConfig": {"temperature": temperature, "maxOutputTokens": max_tokens},
        }

    def _gemini_url(self, stream=False):
        model = urllib.parse.quote(self.gemini_model, safe="")
        method = "streamGenerateContent" if stream else "generateContent"
        suffix = "&alt=sse" if stream else ""
        return f"https://generativelanguage.googleapis.com/v1beta/models/{model}:{method}?key={urllib.parse.quote(self.gemini_key)}{suffix}"

    def _gemini_chat(self, system, messages, temperature=.7, max_tokens=800):
        r = _post_json(self._gemini_url(), self._gemini_payload(system, messages, temperature, max_tokens), timeout=120)
        try:
            return r["candidates"][0]["content"]["parts"][0]["text"].strip()
        except (KeyError, IndexError, TypeError) as exc:
            raise RuntimeError("Gemini returned no text response.") from exc

    def chat(self, system, messages, temperature=.7, max_tokens=800):
        if not self.configured:
            raise RuntimeError("No real LLM provider is configured.")
        if self.gemini_key:
            return self._gemini_chat(system, messages, temperature, max_tokens)
        payload = {"model": self.model, "messages": [{"role": "system", "content": system}] + messages,
                   "temperature": temperature, "max_tokens": max_tokens, "stream": False}
        r = _post_json(f"{self.base_url}/chat/completions", payload, self.api_key, timeout=120)
        return r["choices"][0]["message"]["content"].strip()

    def stream(self, system, messages, temperature=.7, max_tokens=1200):
        if not self.configured:
            raise RuntimeError("No real LLM provider is configured.")
        if self.gemini_key:
            data = json.dumps(self._gemini_payload(system, messages, temperature, max_tokens)).encode("utf-8")
            req = urllib.request.Request(self._gemini_url(stream=True), data=data, method="POST",
                                         headers={"Content-Type": "application/json"})
            with urllib.request.urlopen(req, timeout=120) as resp:
                for raw in resp:
                    line = raw.decode("utf-8", "ignore").strip()
                    if not line.startswith("data:"):
                        continue
                    try:
                        obj = json.loads(line[5:].strip())
                        for part in obj.get("candidates", [{}])[0].get("content", {}).get("parts", []):
                            text = part.get("text")
                            if text:
                                yield text
                    except json.JSONDecodeError:
                        continue
            return

        payload = {"model": self.model, "messages": [{"role": "system", "content": system}] + messages,
                   "temperature": temperature, "max_tokens": max_tokens, "stream": True}
        data = json.dumps(payload).encode("utf-8")
        req = urllib.request.Request(f"{self.base_url}/chat/completions", data=data, method="POST",
                                     headers={"Content-Type": "application/json", "Authorization": f"Bearer {self.api_key}"})
        with urllib.request.urlopen(req, timeout=120) as resp:
            for raw in resp:
                line = raw.decode("utf-8", "ignore").strip()
                if not line.startswith("data:"):
                    continue
                item = line[5:].strip()
                if item == "[DONE]":
                    break
                try:
                    obj = json.loads(item)
                    delta = obj.get("choices", [{}])[0].get("delta", {}).get("content")
                    if delta:
                        yield delta
                except json.JSONDecodeError:
                    continue


class ImageProvider:
    def __init__(self):
        self.base_url = os.getenv("UF_IMAGE_BASE_URL", "").rstrip("/")

    def generate(self, prompt, size="1024x1024"):

        return item.get("url")

"""Safe, fixed-domain web search adapter. No arbitrary URL fetching."""
import html,re,urllib.parse,urllib.request

def search(query,limit=5):
    q=urllib.parse.quote_plus(query[:300]); url='https://html.duckduckgo.com/html/?q='+q
    req=urllib.request.Request(url,headers={'User-Agent':'Aikiyara/1.0'})
    with urllib.request.urlopen(req,timeout=12) as r: text=r.read().decode('utf-8','ignore')
    out=[]
    for m in re.finditer(r'class="result__a"[^>]*href="([^"]+)"[^>]*>(.*?)</a>',text,re.S):
        href=html.unescape(m.group(1)); title=re.sub('<.*?>','',html.unescape(m.group(2))).strip()
        if href.startswith('//'): href='https:'+href
        out.append({'title':title,'url':href})
        if len(out)>=limit: break
    return out

import re
from collections import defaultdict

LEXICON = {
    "breakup": ["breakup", "break up", "ex", "relationship ended", "left me", "ब्रेकअप", "ब्रेक अप"],
    "loss": ["lost", "miss", "missing", "gone", "खो", "याद आ रही", "याद आता"],

    "loneliness": ["lonely", "alone", "no one", "nobody", "अकेला", "अकेली", "कोई नहीं"],
    "confusion": ["confused", "don't know", "do not know", "not sure", "unclear", "समझ नहीं", "पता नहीं"],
    "joy": ["happy", "excited", "glad", "great", "amazing", "खुश", "उत्साहित", "बहुत अच्छा"],

MAPPING = {
    "breakup": ["hurt", "grief", "conflict", "uncertainty"],
    "loss": ["sadness", "grief", "longing"],
    "anger": ["anger", "frustration"],
    "fear": ["fear", "worry", "uncertainty"],

    "joy": ["happiness", "excitement"],
    "sadness": ["sadness", "hurt"],
    "stress": ["stress", "overload", "worry"],
}
NEGATIONS = {"not", "no", "never", "isn't", "wasn't", "don't", "do not", "didn't", "नहीं", "मत"}
INTENSIFIERS = {"very", "really", "so", "extremely", "बहुत", "बहुत ज्यादा", "काफी", "बहुत ही"}


def _contains(text: str, term: str) -> bool:
    if any(ord(c) > 127 for c in term):
        return term.casefold() in text.casefold()
    return bool(re.search(r"(?<!\w)" + re.escape(term) + r"(?!\w)", text, re.I))


def _is_negated(text: str, term: str) -> bool:
    if any(ord(c) > 127 for c in term):
        return False
    match = re.search(r"(?<!\w)" + re.escape(term) + r"(?!\w)", text, re.I)
    if not match:
        return False
    window = text[max(0, match.start() - 28):match.start()].casefold()
    return any(re.search(r"(?<!\w)" + re.escape(n) + r"(?!\w)", window) for n in NEGATIONS)


def _intensity(text: str, hits: list[str]) -> float:
    if not hits:
        return 0.0
    score = 0.5
    if "!" in text or "!!" in text:
        score += 0.1
    if any(_contains(text, x) for x in INTENSIFIERS):
        score += 0.2
    if len(hits) >= 2:
        score += 0.1
    return min(score, 1.0)


def analyze_emotional_signals(message: str, history: list[dict]) -> dict:
    text = message.casefold()
    scores = defaultdict(float)
    observed = []
    for category, words in LEXICON.items():
        hits = [w for w in words if _contains(text, w) and not _is_negated(text, w)]
        if hits:
            intensity = _intensity(text, hits)
            scores[category] += intensity
            observed.append({"category": category, "matched_terms": hits, "intensity": round(intensity, 2)})

    recent = " ".join(str(x.get("content", "")) for x in history[-8:] if x.get("role") == "user").casefold()
    if recent and "breakup" in recent and "breakup" not in scores:
        scores["breakup"] += 0.25
        observed.append({"category": "recent_relationship_context", "matched_terms": ["recent context"], "intensity": 0.25})

    hypotheses = []
    for category, _score in sorted(scores.items(), key=lambda x: x[1], reverse=True):
        for hypothesis in MAPPING.get(category, []):
            if hypothesis not in hypotheses:
                hypotheses.append(hypothesis)

    if not observed:
        confidence = "low"
    elif len(observed) == 1:
        confidence = "medium"
    else:
        confidence = "medium-high"

    primary = max(scores, key=scores.get) if scores else None
    return {
        "observed_signals": observed,
        "hypotheses": hypotheses[:8],
        "confidence": confidence,
        "primary_emotion": primary,
        "emotion_scores": {k: round(min(v, 1.0), 2) for k, v in sorted(scores.items(), key=lambda x: x[1], reverse=True)},
    }

def extract_text(path):
    p=Path(path); ext=p.suffix.lower()
    if ext in {'.txt','.md','.csv','.json','.py','.js','.html','.css'}: return p.read_text(errors='ignore')[:120000]

    raise RuntimeError('Unsupported file type.')

import os
from .llm_adapter import LocalTransformersAdapter
from .provider import OpenAICompatibleProvider

SYSTEM_PROMPT = """You are Aikiyara, a multilingual AI assistant.
You can converse naturally in the user's language or mixed languages. Never claim to read minds; emotional analysis is uncertain evidence.
Be helpful, truthful and concise. Preserve context. If asked for code, provide working code and explain important assumptions. If asked for an image, describe the prompt clearly for the configured image generator. Do not invent memories, tools, files, citations, or completed actions.
"""

_PROVIDER = OpenAICompatibleProvider()
_ADAPTER = None

def _adapter():
    global _ADAPTER
    if _ADAPTER is not None:
        return _ADAPTER
    model_id = os.getenv('UF_MODEL_ID', '').strip()
    if not model_id:
        return None
    try:
        _ADAPTER = LocalTransformersAdapter(model_id)
    except Exception:
        _ADAPTER = None
    return _ADAPTER


def _fallback(message, signals):
    primary = signals.get('primary_emotion')
    if primary in {'fear', 'stress'}:
        return "I may be picking up some worry or pressure. I could be wrong—what part is bothering you most?"
    if primary in {'sadness', 'loss', 'breakup'}:
        return "That sounds emotionally heavy. I don't want to assume exactly how you feel. What happened?"
    if primary == 'anger':
        return "I may be picking up frustration or anger. What happened just before this?"
    if primary == 'loneliness':
        return "It sounds like connection or support may matter here. What would help most right now?"
    if primary == 'joy':
        return "That sounds positive! Tell me what happened."
    return "I'm listening. Tell me a little more and I'll use the context rather than jumping to conclusions."


def generate_response(message, history, signals, memories=None):
    memories = memories or []
    context = SYSTEM_PROMPT + '\nUncertain emotion signals: ' + str(signals)
    if memories:
        context += '\nUser-approved memories (use only when relevant): ' + '; '.join(memories)
    messages = [{'role': x.get('role', 'user'), 'content': x.get('content', '')} for x in history[-16:]]
    messages.append({'role': 'user', 'content': message})

    if _PROVIDER.configured:
        try:
            return _PROVIDER.chat(context, messages)
        except Exception:
            pass
    adapter = _adapter()
    if adapter is not None:
        try:
            return adapter.generate(context, history, message)[:4000]
        except Exception:
            pass
    return _fallback(message, signals)

def relevant_memories(query, memories, limit=8):

    scored=[]

===== FILE: README.md =====
# Aikiyara — AI-first PWA

A clean, mobile-first AI assistant interface with a server-side LLM gateway, conversation history, memory, coding assistance, file understanding, web search, voice input and optional image generation.

## Gemini setup (recommended)
Set these on the server (for example Render Environment Variables):

```text
GEMINI_API_KEY=your_key_here
GEMINI_MODEL=gemini-2.5-flash
```

The browser never receives the Gemini key. The server calls the Gemini REST API.

An OpenAI-compatible fallback is also supported with `UF_LLM_BASE_URL`, `UF_LLM_API_KEY`, and `UF_LLM_MODEL`.

## Run

```bash
pip install -r requirements.txt
python app.py
```

## Real features in this repository
- Native Gemini chat and streaming when `GEMINI_API_KEY` is configured.
- Conversation history stored per browser session in SQLite.
- User-approved memory.
- Text/PDF/DOCX/code file extraction.
- Image understanding when the configured LLM supports multimodal input.
- Coding-assistant endpoint.
- Public web search adapter.
- Browser voice input and text-to-speech for assistant replies.
- PWA installation and offline shell.
- Server rate limiting and security headers.

## Important limitations
This is an application using an existing model API, not a newly trained foundation model. It does not include user accounts, multi-device cloud history, a server-side job scheduler, arbitrary server code execution, a true payment system, or a custom multi-search engine. Image generation requires a separately configured OpenAI-compatible image provider. The UI intentionally hides unavailable image generation instead of showing a dead button.

The app uses cookie-based anonymous sessions. For a production multi-user deployment, add real authentication and a durable user identity/database isolation layer before exposing private data to multiple users.

const ASSETS=['/','/manifest.webmanifest','/static/app.css','/static/app.js','/static/icons/icon-192.png','/static/icons/icon-512.png'];
self.addEventListener('install',event=>event.waitUntil(caches.open(CACHE).then(cache=>cache.addAll(ASSETS)).then(()=>self.skipWaiting())));
self.addEventListener('activate',event=>event.waitUntil(caches.keys().then(keys=>Promise.all(keys.filter(key=>key!==CACHE).map(key=>caches.delete(key)))).then(()=>self.clients.claim())));
self.addEventListener('fetch',event=>{
  if(event.request.method!=='GET') return;
  const url=new URL(event.request.url);
  if(url.origin!==self.location.origin) return;
  // Never cache API responses: chat/history/capabilities must stay fresh.
  if(url.pathname.startsWith('/api/')) return;
  event.respondWith(fetch(event.request).then(response=>{
    if(response.ok){const copy=response.clone();caches.open(CACHE).then(cache=>cache.put(event.request,copy));}
    return response;
  }).catch(()=>caches.match(event.request).then(response=>response||caches.match('/'))));

# Aikiyara — Advanced Roadmap

## Implemented in this build
- PWA install/offline shell
- Secure same-origin chat endpoint
- Request size + rate limiting
- Session-isolated conversation history
- Context-aware emotion scoring with intensity and negation handling
- Hindi/Hinglish lexicon coverage
- Optional local Transformers inference via `UF_MODEL_ID`
- Graceful fallback when no model is configured
- Structured emotion metadata for future training/evaluation

## Next engineering stages
1. Persistent opt-in memory (SQLite first, vector retrieval later)
2. Intent + emotion + conversation-state planner
3. Evaluation dataset for Hindi/Hinglish emotional understanding
4. Retrieval-augmented memory with explicit user controls
5. Model routing (small local model vs remote model)
6. Streaming generation and cancellation
7. Authentication and per-user encrypted storage for production deployment
8. Safety classifier and escalation-aware response policy
9. Automated regression benchmark and load tests
10. Fine-tuning only after a quality dataset and evaluation protocol exist

A foundation model is not included in the repository. `UF_MODEL_ID` connects an existing compatible model; it does not train a new one.
